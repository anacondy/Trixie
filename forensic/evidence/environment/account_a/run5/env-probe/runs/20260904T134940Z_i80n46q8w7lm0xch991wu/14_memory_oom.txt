import subprocess,sys
code='''
import sys
mb=int(sys.argv[1])
try:
    b=bytearray(mb*1024*1024)
    for i in range(0,len(b),4096): b[i]=1

except MemoryError:
    print(f'  {mb} MB: MemoryError (refused)')
'''
for mb in [256,512,1024,1500,1800,2500,4000]:
